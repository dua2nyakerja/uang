📱 Telegram Bot — Setup Gratis Rp 0

          
Kirim foto struk dari Telegram → Bot analisis otomatis → Data masuk ke FinKel. Semua komponen di bawah ini gratis!

          

            ✅ Telegram Bot API — Gratis
            ✅ Cloudflare Workers — Gratis
            ✅ Firebase DB — Gratis 1GB
            ✅ Groq AI Vision — Gratis tier
          


          

            
Langkah 1

            
Buat Bot Telegram

            
1. Buka Telegram → cari @BotFather
2. Ketik /newbot
3. Beri nama: FinKel Bareza
4. Beri username: finkelbareza_bot
5. Salin TOKEN yang diberikan

            
Contoh token: 7441234567:AAFxxxxxxxxxxxxxxxxxxxxxxxxxxxx

          


          

            
Langkah 2

            
Daftar Groq (AI analisis foto struk — gratis)

            
1. Buka console.groq.com
2. Sign up gratis (bisa pakai Google)
3. Klik API Keys → Create API Key
4. Salin API Key-nya

            
Contoh: gsk_xxxxxxxxxxxxxxxxxxxxxxxxxx

          


          

            
Langkah 3

            
Setup Firebase Database (gratis 1GB)

            
1. Buka console.firebase.google.com
2. Create project → nama: finkel-bareza
3. Build → Realtime Database → Create database
4. Pilih Test mode
5. Salin URL database

            
Contoh URL: https://finkel-bareza-default-rtdb.firebaseio.com

          


          

            
Langkah 4

            
Deploy ke Cloudflare Workers (gratis, 100K req/hari)

            
1. Buka workers.cloudflare.com → daftar gratis
2. Klik Create Worker
3. Hapus kode default, paste kode di bawah
4. Ganti 3 variabel di bagian atas
5. Klik Deploy
6. Salin URL worker (mis: finkel.namauser.workers.dev)

            
const TELEGRAM_TOKEN = "TOKEN_BOT_KAMU";
const GROQ_KEY       = "GROQ_API_KEY_KAMU";
const FIREBASE_URL   = "https://finkel-bareza-xxxxx.firebaseio.com";

export default {
  async fetch(req) {
    const body = await req.json().catch(()=>({}));
    const msg  = body.message;
    if (!msg) return new Response("ok");
    const chatId = msg.chat.id;

    // ── FOTO STRUK ──
    if (msg.photo) {
      const ph  = msg.photo[msg.photo.length - 1];
      const fR  = await fetch(`https://api.telegram.org/bot${TELEGRAM_TOKEN}/getFile?file_id=${ph.file_id}`);
      const fD  = await fR.json();
      const url = `https://api.telegram.org/file/bot${TELEGRAM_TOKEN}/${fD.result.file_path}`;

      const ai  = await fetch("https://api.groq.com/openai/v1/chat/completions", {
        method: "POST",
        headers: {"Authorization":`Bearer ${GROQ_KEY}`,"Content-Type":"application/json"},
        body: JSON.stringify({
          model: "llama-3.2-11b-vision-preview",
          messages: [{role:"user",content:[
            {type:"text", text:"Analisis struk ini. Balas HANYA JSON tanpa markdown: {tanggal, waktu, toko, items:[{nama,harga}], total}"},
            {type:"image_url", image_url:{url}}
          ]}],
          max_tokens: 600
        })
      });
      const ad  = await ai.json();
      let res;
      try { res = JSON.parse(ad.choices[0].message.content); }
      catch { res = {toko:"?", items:[], total:0, tanggal: new Date().toISOString().split("T")[0], waktu:""}; }

      const ts = Date.now();
      await fetch(`${FIREBASE_URL}/expenses/${ts}.json`, {
        method:"PUT",
        body:JSON.stringify({
          id:ts, category:"Belanja Barang",
          amount: res.total||0,
          date: res.tanggal || new Date().toISOString().split("T")[0],
          notes:`${res.toko||"?"} | ${(res.items||[]).map(i=>i.nama).join(", ")}`,
          items: res.items||[], source:"telegram"
        })
      });

      const reply = [
        `✅ Struk dianalisis!`,
        `🏪 Toko: ${res.toko||"-"}`,
        `📅 ${res.tanggal||"-"} ${res.waktu||""}`,
        ...(res.items||[]).map(i=>`  • ${i.nama}: Rp ${(i.harga||0).toLocaleString("id")}`),
        ``,`💰 Total: Rp ${(res.total||0).toLocaleString("id")}`,
        `✅ Tercatat di FinKel!`
      ].join("\n");

      await fetch(`https://api.telegram.org/bot${TELEGRAM_TOKEN}/sendMessage`,{
        method:"POST", headers:{"Content-Type":"application/json"},
        body:JSON.stringify({chat_id:chatId, text:reply})
      });
    }

    // ── TEKS MANUAL ──
    const txt = msg.text||"";
    if (txt.startsWith("/exp ")) {
      const [,raw,...rest] = txt.split(" ");
      const amount = parseInt(raw)||0; const note = rest.join(" ");
      const ts = Date.now();
      await fetch(`${FIREBASE_URL}/expenses/${ts}.json`,{method:"PUT",body:JSON.stringify({id:ts,category:"Lainnya",amount,date:new Date().toISOString().split("T")[0],notes:note,source:"telegram-manual"})});
      await fetch(`https://api.telegram.org/bot${TELEGRAM_TOKEN}/sendMessage`,{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({chat_id:chatId,text:`✅ Pengeluaran Rp ${amount.toLocaleString("id")} (${note}) dicatat!`})});
    }
    if (txt.startsWith("/inc ")) {
      const [,raw,...rest] = txt.split(" ");
      const amount = parseInt(raw)||0; const note = rest.join(" ");
      const ts = Date.now();
      await fetch(`${FIREBASE_URL}/income/${ts}.json`,{method:"PUT",body:JSON.stringify({id:ts,category:"extra",amount,date:new Date().toISOString().split("T")[0],notes:note,source:"telegram-manual"})});
      await fetch(`https://api.telegram.org/bot${TELEGRAM_TOKEN}/sendMessage`,{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({chat_id:chatId,text:`✅ Pemasukan Rp ${amount.toLocaleString("id")} (${note}) dicatat!`})});
    }
    return new Response("ok");
  }
};
